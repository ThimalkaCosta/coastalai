def _safe_val(v):
    """Make a single Python/numpy value JSON-safe."""
    if v is None:
        return None
    if isinstance(v, float) and (_math.isnan(v) or _math.isinf(v)):
        return None
    if hasattr(v, 'item'):          # numpy scalar
        return _safe_val(v.item())
    if isinstance(v, (bool, int, float, str)):
        return v
    return None
